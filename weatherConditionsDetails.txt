Title- Weather conditions
Readme.md / Description-

A weather prediction and monitoring web application that provides real-time weather information based on user location or city search. The application displays important weather parameters such as temperature, humidity, wind speed, and atmospheric conditions using an interactive and user-friendly interface.

The project helps users quickly access current weather conditions and demonstrates the integration of APIs, frontend development, and real-time data handling in a web application environment.

Dataset / API Used-
The project uses real-time weather data from weather APIs.

const apiKey = '736226bbca25292fb022b7d8e8e420a3';

async function getWeather() {
    console.log("Search button clicked!");
    const city = document.getElementById('cityInput').value;
    const weatherUrl = `https://api.openweathermap.org/data/2.5/weather?q=${city}&units=metric&appid=${apiKey}`;
    const forecastUrl = `https://api.openweathermap.org/data/2.5/forecast?q=${city}&units=metric&appid=${apiKey}`;

    try {
        // Fetch Current Weather
        const weatherRes = await fetch(weatherUrl);
        const weatherData = await weatherRes.json();
        
        // Update Dashboard
        document.getElementById('cityName').innerText = weatherData.name;
        document.getElementById('temp').innerText = `${Math.round(weatherData.main.temp)}°C`;
        document.getElementById('condition').innerText = weatherData.weather[0].main;
        document.getElementById('humidity').innerText = `Humidity: ${weatherData.main.humidity}%`;
        document.getElementById('wind').innerText = `Wind: ${weatherData.wind.speed} m/s`;
        
        // Dynamic Background
        document.body.className = `bg-${weatherData.weather[0].main}`;
        document.getElementById('dashboard').classList.remove('hidden');

        // Fetch Forecast
        const forecastRes = await fetch(forecastUrl);
        const forecastData = await forecastRes.json();
        
        // Filter for one reading per day (roughly)
        const forecastHTML = forecastData.list
            .filter((_, index) => index % 8 === 0)
            .map(day => `<div><p>${day.dt_txt.slice(5, 10)}</p><p>${Math.round(day.main.temp)}°C</p></div>`)
            .join('');
        
        document.getElementById('forecast').innerHTML = forecastHTML;

    } catch (error) {
        alert("City not found or API error.");
    }
}
